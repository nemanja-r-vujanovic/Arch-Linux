vim "file.txt"     = Open file
i                  = Insert mode
Esc                = Normal mode
:saveas "file.txt" = Save as
:w                 = Save
:wq                = Close with saving
:q!                = Close without saving
u                  = Undo
Ctrl+r             = Redo
/word              = Find case-sensitive (n = next; N = previous)
/word\c            = Find case-insensitive (n = next; N = previous)
y                  = Copy
2yy                = Copy 2 lines
:2,5y              = Copy from line 2 to line 5 (inclusive)
d                  = Cut
2dd                = Cut 2 lines
:2,5d              = Cut from line 2 to line 5 (inclusive)
p                  = Paste
x                  = Delete char
D                  = Delete line
v                  = Select char
V                  = Select lines
Ctrl+v             = Select rectangular
ggVG               = Select all
gg                 = Jump to first line
G                  = Jump to last line
:123               = Jump to line 123
w                  = Move to next word
b                  = Move to previous word
:vsplit "file.txt" = Side-by-side file (Ctrl + W, H = Left; Ctrl + W, L = Right; Ctrl + W, Q = Close)
:term              = Terminal bottom (Ctrl + W, Shift + J = Top)
~                  = Toggle case of selected char
:%s/old/new/       = Replace first occurrence
:%s/old/new/g      = Replace all occurrences
:sort              = Sort selected
:sort!             = Reverse sort selected
:%sort             = Sort entire file
:%sort!            = Reverse sort entire file
